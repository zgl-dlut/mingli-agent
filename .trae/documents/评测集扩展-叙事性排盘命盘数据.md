def check_structure(report: dict, user_question: str) -> list[dict]:
    """检查报告结构合规性，返回违规列表"""
    violations = []
    
    # 1. 子问题拆解覆盖度
    #    估算用户问题中的独立关切点数量，与 subtopics 数量比对
    expected_count = estimate_question_count(user_question)  # 简单分词/标点切分
    actual_count = len(report.get("subtopics", []))
    if actual_count < expected_count:
        violations.append({
            "type": "子问题漏答",
            "detail": f"用户问了{expected_count}个独立问题，只拆了{actual_count}个子问题"
        })
    
    # 2. 每个 subtopic 的 blocks 数量（2-4）
    for i, sub in enumerate(report.get("subtopics", [])):
        blocks = sub.get("blocks", [])
        if len(blocks) < 2 or len(blocks) > 4:
            violations.append({
                "type": "block数量异常",
                "subtopic": i,
                "detail": f"blocks数量={len(blocks)}，应为2-4"
            })
    
    # 3. 每个 block 的 keyword 非空、narrative 字数 30-120
    for i, sub in enumerate(report.get("subtopics", [])):
        for j, blk in enumerate(sub.get("blocks", [])):
            kw = blk.get("keyword", "")
            nar = blk.get("narrative", "")
            if not kw.strip():
                violations.append({"type": "keyword为空", "subtopic": i, "block": j})
            if len(nar) < 30:
                violations.append({"type": "narrative过短", "subtopic": i, "block": j, "len": len(nar)})
            if len(nar) > 120:
                violations.append({"type": "narrative过长", "subtopic": i, "block": j, "len": len(nar)})
    
    # 4. summary items 的 keyword 叙事化（非泛泛标签）
    for item in report.get("summary", {}).get("items", []):
        kw = item.get("keyword", "")
        # 检查是否为"蹲守期/守业期/蛰伏期"等泛泛标签
        if re.match(r"^[蛰伏蹲守守业过渡调整观望]+期$", kw):
            violations.append({"type": "summary关键词泛泛", "keyword": kw})
    
    return violations
```

def check_narrative(report: dict) -> list[dict]:
    """检查叙事性合规性"""
    violations = []
    
    # 1. keyword 是否含术语黑名单词
    TERM_BLACKLIST = ["七杀", "正官", "偏印", "正印", "伤官", "食神", 
                      "比肩", "劫财", "偏财", "正财", "月令", "宫位", "用神", "忌神"]
    for i, sub in enumerate(report.get("subtopics", [])):
        for j, blk in enumerate(sub.get("blocks", [])):
            kw = blk.get("keyword", "")
            for term in TERM_BLACKLIST:
                if term in kw:
                    violations.append({
                        "type": "keyword含术语",
                        "subtopic": i, "block": j,
                        "keyword": kw, "term": term
                    })
    
    # 2. narrative 中术语是否有解释（括号/= /也就是）
    #    术语出现后 10 字符内必须有解释标记
    for i, sub in enumerate(report.get("subtopics", [])):
        for j, blk in enumerate(sub.get("blocks", [])):
            nar = blk.get("narrative", "")
            for term in TERM_BLACKLIST:
                idx = nar.find(term)
                if idx >= 0:
                    after = nar[idx+len(term):idx+len(term)+10]
                    if not any(m in after for m in ["（", "(", "=", "也就是", "即"]):
                        violations.append({
                            "type": "术语无解释",
                            "subtopic": i, "block": j,
                            "term": term
                        })
    
    return violations
```

def check_chart_correctness(generated: dict, expected: dict) -> list[dict]:
    """比对排盘结果与标准答案"""
    violations = []
    
    # 1. 四柱干支
    gen_pillars = generated.get("四柱", "").replace(" ", "")
    exp_pillars = expected.get("四柱", "").replace(" ", "")
    if gen_pillars != exp_pillars:
        violations.append({
            "type": "四柱错误",
            "expected": exp_pillars,
            "actual": gen_pillars
        })
    
    # 2. 十神（已知存在阴干日主正偏全反的 bug）
    gen_shishen = generated.get("十神", {})
    exp_shishen = expected.get("十神", {})
    for pos, exp_val in exp_shishen.items():
        gen_val = gen_shishen.get(pos, "")
        if gen_val != exp_val:
            violations.append({
                "type": "十神错误",
                "position": pos,
                "expected": exp_val,
                "actual": gen_val
            })
    
    # 3. 大运方向
    gen_dir = generated.get("大运方向", "")
    exp_dir = expected.get("大运方向", "")
    if gen_dir != exp_dir:
        violations.append({"type": "大运方向错误", "expected": exp_dir, "actual": gen_dir})
    
    return violations
```

def extract_chart_entities(report: dict) -> dict:
    """从 LLM 输出中提取所有命理相关实体"""
    entities = {"大运": set(), "流年": set(), "十神": set(), "用神": set(), "忌神": set()}
    
    # 大运：匹配 干支+运（如"甲寅大运"）
    # 流年：匹配 数字+年+干支（如"2034年甲寅"）
    # 十神：匹配术语黑名单词
    # 用神/忌神：匹配"用神""忌神"后的五行字
    
    all_text = json.dumps(report, ensure_ascii=False)
    
    # 大运干支
    for m in re.finditer(r"([甲乙丙丁戊己庚辛壬癸][子丑寅卯辰巳午未申酉戌亥])大运", all_text):
        entities["大运"].add(m.group(1))
    
    # 流年干支
    for m in re.finditer(r"(\d{4})年([甲乙丙丁戊己庚辛壬癸][子丑寅卯辰巳午未申酉戌亥])", all_text):
        entities["流年"].add(f"{m.group(1)}:{m.group(2)}")
    
    # 十神（黑名单术语）
    shishen_terms = ["七杀", "正官", "偏印", "正印", "伤官", "食神", "比肩", "劫财", "偏财", "正财"]
    for term in shishen_terms:
        if term in all_text:
            entities["十神"].add(term)
    
    return entities


def check_chart_usage(report: dict, chart_data: dict) -> list[dict]:
    """检查 LLM 输出是否编造了命盘未提供的数据"""
    violations = []
    entities = extract_chart_entities(report)
    
    # 1. 大运：LLM 提到的大运干支必须在命盘大运列表中
    chart_dayun = set(extract_dayun_from_chart(chart_data))  # 从命盘提取所有大运干支
    for dy in entities["大运"]:
        if dy not in chart_dayun:
            violations.append({"type": "编造大运", "entity": dy})
    
    # 2. 流年：LLM 提到的流年干支必须在重点流年列表中
    chart_liunian = set(extract_liunian_from_chart(chart_data))
    for ln in entities["流年"]:
        year, gz = ln.split(":")
        if ln not in chart_liunian and year not in [l.split(":")[0] for l in chart_liunian]:
            violations.append({"type": "编造流年", "entity": ln})
    
    # 3. 用神/忌神：与命盘一致
    chart_yongshen = chart_data.get("用神", "")
    chart_jishen = chart_data.get("忌神", "")
    for ys in entities["用神"]:
        if ys not in chart_yongshen:
            violations.append({"type": "用神不一致", "expected": chart_yongshen, "actual": ys})
    
    return violations
```

## Assumptions & Decisions

1. **代码层检查优先**：结构检查、叙事性检查、命盘数据比对用代码实现（快、可重复），LLM 评分作为补充（慢、有随机性）

2. **LLM 评分用 qwen3.8-flash**：评分任务简单，用轻量模型节省成本

3. **排盘标准答案手工构造**：20-30 个覆盖关键场景的命盘，用权威排盘工具（如元亨利贞）生成标准答案

4. **命盘实体提取用正则**：大运/流年干支有固定格式（天干+地支），正则可靠提取

5. **评测集不入库**：JSON 文件方便版本管理和 diff

6. **chart_usage_check 不强制拦截**：命盘数据可能有叙事化转述，只告警不拦截，避免误杀
